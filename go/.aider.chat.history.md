main.go

```
package main

import (
	"flag"
	"fmt"
	"log"

	"our-package-manager/githelper"
	"our-package-manager/makehelper"
)

func main() {
	repoURL := flag.String("repo", "", "URL of the Git repository")
	target := flag.String("target", "install", "Makefile target to execute")
	flag.Parse()

	if *repoURL == "" {
		log.Fatal("Repository URL must be provided")
	}

	gitRepo := githelper.NewGitRepository(*repoURL)

	err := gitRepo.Clone(1, "")
	if err != nil {
		log.Fatalf("Failed to clone repository: %s", err)
	}

	err = makehelper.MakeTarget(gitRepo.Directory, "Makefile", *target)
	if err != nil {
		log.Fatalf("Failed to execute make target: %s", err)
	}

	fmt.Println("Installation completed successfully.")
}
```

This implementation allows the user to specify a Git repository URL and a target to execute from the Makefile. The program will clone the repository and run the specified target, which defaults to "install".

To create a package manager that installs applications from a Git repository according to a Makefile, I will implement a Python script that performs the following steps:

1. Clone the Git repository.
2. Navigate into the cloned directory.
3. Execute the `make` command to build/install the application.

Here is the implementation:

package_manager.py
```
import os
import subprocess
import sys
import git

def install_from_git(repo_url):
    # Clone the repository
    repo_name = repo_url.split('/')[-1].replace('.git', '')
    if not os.path.exists(repo_name):
        print(f"Cloning repository {repo_url}...")
        git.Repo.clone_from(repo_url, repo_name)
    else:
        print(f"Repository {repo_name} already exists. Pulling latest changes...")
        repo = git.Repo(repo_name)
        repo.remotes.origin.pull()

    # Change directory to the cloned repository
    os.chdir(repo_name)

    # Check if Makefile exists
    if os.path.isfile('Makefile'):
        print("Found Makefile. Running 'make'...")
        subprocess.run(['make'], check=True)
    else:
        print("No Makefile found in the repository.")

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print("Usage: python package_manager.py <git-repo-url>")
        sys.exit(1)

    install_from_git(sys.argv[1])
```
